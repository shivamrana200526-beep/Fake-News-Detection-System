A. Server Setup & Middleware Configuration — artifacts/api-server/src/app.js

Shivam architected the Express instance with robust JSON parsing to support high-resolution base64 screenshot uploads:

// Author: Shivam (Backend Lead)
// File: artifacts/api-server/src/app.js
import express from "express";
import cors from "cors";
import pinoHttp from "pino-http";
import router from "./routes/index.js";
import { logger } from "./lib/logger.js";

const app = express();

// High-performance structured JSON logging
app.use(pinoHttp({ logger }));

// CORS configured to permit cross-origin calls from Vite dev server and production domains
app.use(cors());

// Parse incoming request bodies up to 50MB (required for base64 image analysis)
app.use(express.json({ limit: "50mb" }));
app.use(express.urlencoded({ extended: true, limit: "50mb" }));

// Route dispatch: Mount all API endpoints under /api
app.use("/api", router);

export default app;

B. Content Analysis & Ingestion Pipeline — artifacts/api-server/src/routes/analyze.js

Shivam created the primary fact-checking endpoint that intercepts URLs for live scraping before handing off to the AI service:

// Author: Shivam (Backend Lead)
// File: artifacts/api-server/src/routes/analyze.js
import { Router } from "express";
import { scrapeUrl } from "../lib/scraper.js";
import { queryAI, runHeuristicFactCheck } from "../lib/ai-service.js";
import { db } from "../lib/db.js"; // Falls back to local storage if DB is offline

const router = Router();

router.post("/analyze", async (req, res) => {
  const { content, type, imageBase64 } = req.body;
  if (!content || !type) {
    return res.status(400).json({ message: "content and type are required" });
  }

  try {
    // 1. If type is URL, extract article text and publication metadata
    let articleContext;
    if (type === "url") {
      const scraped = await scrapeUrl(content);
      if (scraped.scrapedOk) {
        articleContext = [
          `Title: ${scraped.title}`,
          `Author: ${scraped.author}`,
          `Published: ${scraped.publishDate}`,
          `Body (${scraped.wordCount} words):\n${scraped.text.substring(0, 4000)}`,
        ].join("\n");
      }
    }

    // 2. Invoke multi-tier AI inference
    const result = await runAnalysis(content, type, imageBase64, articleContext);

    // 3. Persist record to persistent storage (PostgreSQL or local JSON storage)
    const savedRecord = await db.saveAnalysis({
      content: type === "image" ? "[Image Upload]" : content,
      sourceType: type,
      prediction: result?.prediction || "Unknown",
      confidence: Number(result?.confidence || 50),
      explanation: result?.explanation || "",
      keywords: result?.keywords || [],
      detectedSource: result?.detectedSource || null,
      factBreakdown: result?.factBreakdown || null,
    });

    return res.json({ ...savedRecord, ...result });
  } catch (err) {
    req.log?.error(err, "Analysis failed");
    return res.status(500).json({ message: "Failed to analyze content" });
  }
});

export default router;

C. Server-Sent Events (SSE) Streaming Engine — artifacts/api-server/src/routes/chat.js

Shivam engineered the streaming chat endpoint providing real-time progressive response tokens:

// Author: Shivam (Backend Lead)
// File: artifacts/api-server/src/routes/chat.js
router.post("/chat", async (req, res) => {
  const { message, conversationId } = req.body;

  // Establish Server-Sent Events protocol headers
  res.setHeader("Content-Type", "text/event-stream");
  res.setHeader("Cache-Control", "no-cache");
  res.setHeader("Connection", "keep-alive");

  const send = (event, data) =>
    res.write(`event: ${event}\ndata: ${JSON.stringify(data)}\n\n`);

  try {
    // 1. Check for domain guardrails (e.g. food recipes / off-topic content)
    if (isRecipeOrOffTopic(message)) {
      send("status", { type: "guardrail", message: "Redirecting query..." });
      send("token", { token: "🛡️ **SatyaCheck Forensic Notice:**\nI am exclusively designed for forensic news verification. I do not provide cooking recipes or unrelated lifestyle content. Please submit news claims or viral headlines for fact-checking!" });
      send("done", {});
      return res.end();
    }

    // 2. Stream tokens in real-time
    send("status", { type: "analyzing", message: "Analyzing query..." });
    await streamChatResponse(message, (token) => {
      send("token", { token });
    });

    send("done", {});
    res.end();
  } catch (err) {
    send("error", { message: "Internal chat processing error" });
    res.end();
  }
});
